1. Buat database
create database perpus82;

2. Create Table
categories
CREATE TABLE categories( id integer PRIMARY KEY AUTO_INCREMENT, name varchar(255) );

books
CREATE TABLE books( id integer PRIMARY KEY AUTO_INCREMENT, title varchar(255), summary text, stock integer, category_id integer, FOREIGN KEY (category_id) REFERENCES categories(id) );
